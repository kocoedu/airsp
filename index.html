    let video = document.getElementById('webcam');
    let canvas = document.getElementById('canvas');
    let isDetecting = false;
    let isProcessing = false;

    async function startCamera() {
      const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user' } });
      video.srcObject = stream;
    }

    function stopCamera() {
      stopDetection();
      if (video.srcObject) {
        video.srcObject.getTracks().forEach(track => track.stop());
        video.srcObject = null;
      }
    }

    function startDetection() {
      if (isDetecting) return;
      isDetecting = true;
      detectLoop();
    }

    function stopDetection() {
      isDetecting = false;
    }

    async function detectLoop() {
      if (!isDetecting) return;
      if (!isProcessing && video.readyState === 4) {
        isProcessing = true;
        try {
          canvas.width = video.videoWidth;
          canvas.height = video.videoHeight;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(video, 0, 0);
          const base64Image = canvas.toDataURL('image/jpeg', 0.8);

          const res = await fetch('/api/predict', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ image: base64Image })
          });
          const data = await res.json();
          updateUI(data);
        } catch (err) {
          console.error(err);
        } finally {
          isProcessing = false;
        }
      }
      setTimeout(detectLoop, 1000); // 약 1초 간격
    }

    function updateUI(data) {
      if (!data || !data.topClass) return;
      const emojiMap = { rock: '✊ 바위', paper: '🖐 보', scissors: '✌️ 가위' };
      document.getElementById('result').innerText = emojiMap[data.topClass] || data.topClass;
      document.getElementById('confidence').innerText = (data.confidence * 100).toFixed(1) + '%';
    }
  </script>

// api/predict.js (Vercel Serverless Function)
// Roboflow Cloud API Key와 Endpoint는 브라우저에 노출하지 않고 서버리스 함수에서 안전하게 호출합니다.

export default async function handler(req, res) {
  // CORS 및 HTTP 메소드 검증
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method Not Allowed' });
  }

  try {
    const { image } = req.body;
    if (!image) {
      return res.status(400).json({ error: '이미지 데이터가 누락되었습니다.' });
    }

    const apiKey = process.env.ROBOFLOW_API_KEY;
    const endpoint = process.env.ROBOFLOW_ENDPOINT;

    if (!apiKey || !endpoint) {
      return res.status(500).json({
        error: 'Vercel 환경변수(ROBOFLOW_API_KEY, ROBOFLOW_ENDPOINT)가 설정되지 않았습니다.'
      });
    }

    // Base64 프리픽스 제거 (data:image/jpeg;base64, ...)
    const base64Data = image.replace(/^data:image\/(png|jpeg|jpg);base64,/, '');

    // Endpoint URL에 api_key 쿼리 파라미터 결합
    let targetUrl = endpoint;
    const separator = targetUrl.includes('?') ? '&' : '?';
    if (!targetUrl.includes('api_key=')) {
      targetUrl = `${targetUrl}${separator}api_key=${apiKey}`;
    }

    // Roboflow Workflow / Inference API 호출 (base64 raw text/plain 전송)
    const response = await fetch(targetUrl, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: base64Data
    });

    if (!response.ok) {
      const errorText = await response.text();
      return res.status(response.status).json({
        error: 'Roboflow API 호출 실패',
        details: errorText
      });
    }

    const resultData = await response.json();

    // Roboflow 결과 파싱 (분류 모델 및 Workflow 응답 구조 정규화)
    let topClass = 'unknown';
    let confidence = 0.0;

    // 1. 분류(Classification) 예측 응답 처리
    if (resultData.predictions && Array.isArray(resultData.predictions) && resultData.predictions.length > 0) {
      topClass = resultData.predictions[0].class || resultData.predictions[0].label;
      confidence = resultData.predictions[0].confidence || 0.0;
    } else if (resultData.predictions && typeof resultData.predictions === 'object') {
      // { predictions: { rock: { confidence: 0.95 }, paper: ... } } 형태
      let maxConf = -1;
      for (const [cls, info] of Object.entries(resultData.predictions)) {
        const conf = info.confidence || info;
        if (typeof conf === 'number' && conf > maxConf) {
          maxConf = conf;
          topClass = cls;
          confidence = conf;
        }
      }
    } else if (resultData.top) {
      topClass = resultData.top;
      confidence = resultData.confidence || 0.0;
    }

    return res.status(200).json({
      success: true,
      topClass: topClass.toLowerCase(),
      confidence: confidence,
      raw: resultData
    });

  } catch (error) {
    console.error('Prediction API Server Error:', error);
    return res.status(500).json({
      error: '서버 내부 오류가 발생했습니다.',
      message: error.message
    });
  }
}

Vercel 배포 및 환경변수 설정 가이드

초보자도 따라 할 수 있는 GitHub 업로드 및 Vercel 배포 단계별 설명서입니다.

1 GitHub 저장소(Repository) 생성 및 업로드
  1. GitHub.com 로그인 후 [New Repository] 클릭
  2. 저장소 이름 작성 (예: rock-paper-scissors-ai) -> Create Repository
  3. 컴퓨터의 프로젝트 폴더에서 아래 명령어 실행하여 저장소 업로드:
git init
git add .
git commit -m "Initial commit"
git branch -M main
git remote add origin https://github.com/사용자이름/저장소이름.git
git push -u origin main
2 Vercel 환경변수(Environment Variables) 설정

API Key가 브라우저에 노출되지 않도록 Vercel 대시보드에서 설정합니다.

ROBOFLOW_API_KEY

Roboflow 계정 settings -> API Key 값

예: rf_xxxxxxxxxKey
ROBOFLOW_ENDPOINT

DINOv3 Workflow URL 또는 Model Endpoint

https://detect.roboflow.com/rock-paper-scissors/1
3 Vercel 프로젝트 가져오기 및 배포
  1. Vercel 대시보드 접속 후 [Add New...] -> [Project] 선택
  2. GitHub 계정을 연결하고 방금 만든 저장소 선택 (Import)
  3. Environment Variables 세션 클릭 후 위의 환경변수 2개 입력
  4. [Deploy] 버튼 클릭 -> 1분 이내 배포 완료!
4 카메라 권한 및 테스트 주의사항
  • HTTPS 필수: 웹캠(getUserMedia)은 Vercel의 보안 HTTPS 프로토콜에서만 정상 동작합니다.
  • 모바일 권한: 스마트폰 브라우저에서 접속 시 카메라 허용 팝업에서 '허용'을 선택하세요.
  • 카메라 차단 시: 우측의 '사진 업로드 인식' 기능을 이용해 가위바위보 손 이미지 분석이 가능합니다.
